#include <stdio.h>
#include <omp.h>

#define N 1000000
#define STRIP_SIZE 8

int main()
{
    static double A[N];
    static double B[N];
    static double C[N];

    for (int i = 0; i < N; i++) {
        A[i] = i * 1.0;
        B[i] = 2.0;
    }

    double tstart = omp_get_wtime();

    #pragma omp parallel for
    for (int strip = 0; strip < N; strip += STRIP_SIZE)
    {
        int end = strip + STRIP_SIZE;

        if (end > N)
            end = N;

        #pragma omp simd
        for (int i = strip; i < end; i++)
        {
            C[i] = A[i] * B[i];
        }
    }

    double tstop = omp_get_wtime();

    printf("Execution time = %f seconds\n",
           tstop - tstart);

    printf("Example result: C[100] = %f\n", C[100]);

    return 0;
}
